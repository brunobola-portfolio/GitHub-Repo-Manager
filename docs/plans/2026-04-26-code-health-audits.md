Create `.dev/audit-ds-classes.mjs`:

```js
// One-off audit: enumerate every .ds-XXX class defined in
// src/design-system.css and report which have zero usages
// elsewhere in the codebase. Output is a list of orphan class
// names — the engineer manually confirms each before removal
// (some classes are referenced by other CSS rules via animation-name
// or @apply, which this grep doesn't catch).

import { readFileSync } from 'node:fs'
import { execSync } from 'node:child_process'

const css = readFileSync('src/design-system.css', 'utf8')

// Match .ds-XXX selectors at the start of a CSS rule.
// The \w-_ class allows hyphens and underscores in class names.
const classes = [...new Set(
    [...css.matchAll(/^\.(ds-[\w-]+)/gm)].map((m) => m[1])
)].sort()

console.log(`Found ${classes.length} ds-* classes defined in design-system.css`)
console.log('')

const orphans = []
const used = []
for (const cls of classes) {
    // Search src/ + docs/ for the class name with word boundaries to avoid
    // false matches on substrings like 'ds-card-shimmer-2'.
    let matches = ''
    try {
        matches = execSync(
            `grep -rn --include="*.jsx" --include="*.js" --include="*.css" --include="*.md" "\\b${cls}\\b" src/ docs/ 2>/dev/null`,
            { encoding: 'utf8' },
        )
    } catch {
        // grep exits 1 when there are no matches — that's the orphan signal.
        matches = ''
    }
    // Filter out the definition file itself.
    const lines = matches
        .split('\n')
        .filter((l) => l && !l.includes('design-system.css'))

    // Also check whether the class is referenced WITHIN design-system.css
    // by another rule (animation-name, @apply, etc.). If yes, not orphan.
    const cssInternalUse = new RegExp(
        `(animation(?:-name)?:[^;]*\\b${cls}\\b|@apply[^;]*\\b${cls}\\b)`,
    ).test(css)

    if (lines.length === 0 && !cssInternalUse) {
        orphans.push(cls)
    } else {
        used.push({ cls, count: lines.length, cssInternal: cssInternalUse })
    }
}

console.log(`Orphans (${orphans.length}):`)
orphans.forEach((c) => console.log(`  - ${c}`))
console.log('')
console.log(`Used (${used.length}):`)
used.forEach(({ cls, count, cssInternal }) => {
    const note = cssInternal ? ' [CSS-internal ref]' : ''
    console.log(`  - ${cls}: ${count} matches${note}`)
})
```

- [ ] **Step 3: Manually verify each orphan**

```bash

grep -rn "ds-CLS" . --include='*.jsx' --include='*.js' --include='*.css' --include='*.md' --include='*.html' 2>/dev/null
```

If the manual grep returns matches the script missed (e.g., classes built from string concatenation, or dynamic className templates the script can't see), keep the class. The script is conservative but not infallible.

If the manual grep confirms zero usages, the class is safe to remove.
